async function renderPdfToCanvas(url, canvasId) {
  try {
    // 1. 加载PDF文档
    const pdfDoc = await pdfjsLib.getDocument(url).promise;
    
    // 2. 获取目标canvas元素
    const canvas = document.getElementById(canvasId);
    if (!canvas) {
      throw new Error(`找不到ID为 ${canvasId} 的canvas元素`);
    }
    const context = canvas.getContext('2d');
    
    // 3. 获取并渲染第一页
    const page = await pdfDoc.getPage(1);
    const viewport = page.getViewport({ scale: 1 });
    
    // 调整canvas尺寸
    canvas.height = viewport.height;
    canvas.width = viewport.width;
    
    // 4. 执行渲染
    await page.render({
      canvasContext: context,
      viewport: viewport
    }).promise;

    console.log(`PDF渲染完成 (${canvasId})`);
    
  } catch (error) {
    console.error('PDF渲染失败:', error);
    throw error; // 可以选择重新抛出或处理错误
  }
}


renderPdfToCanvas('https://github.com/xwy231321/xwy/rew/main/试卷/A大一/大学英语/大学英语1复习.pdf', 'pdf-canvas2')
  .then(() => console.log('渲染完成'))
  .catch(err => console.error('发生错误:', err));
</script>
